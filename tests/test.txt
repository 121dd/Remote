#include <atlimage.h>
#include <ShellScalingApi.h>
int HandleScreen(Packet* pck){
    //获取本地屏幕
    //1.创建一个image对象,创建一个画布
    CImage image;
    //2.拿到屏幕上下文，获取许可证
    HDC hScreen = GetDC(NULL);
    //3.拿到屏幕像素位宽，获取深度
    int bitWidth = GetDeviceCaps(hScreen, BITSPIXEL);
    std::cout << "bitWidth:" << bitWidth << std::endl;

    SetProcessDpiAwarenessContext()//获取全部的屏幕，否则拿到的是逻辑上的，不真实的
    //4.获取屏幕的宽高（以像素为单位）
    int sWidth = GetSystemMetrics(SM_CXSCREEN);
    int sHeight = GetSystemMetrics(SM_CYSCREEN);
    image.Create(sWidth, sHeight, bitWidth);
    //5.把屏幕数据复制到image里面
    BitBlt(image.GetDC(), 0, 0, sWidth, sHeight, hScreen, 0, 0, SRCCOPY); // 执行真正的截图操作，把屏幕内容复制到 image 中
    //释放屏幕
    ReleaseDC(NULL, hScreen);

    //6.转为一个网络流
    //从堆上申请一块可变化的空间
    HGLOBAL hMen = GlobalAlloc(GMEM_MOVEABLE, 0);
    if(hMen == NULL){
        //分配失败
        return -1;
    }

    //创建一个内存
    IStream* pStream = NULL;
    HRESULT ret =  CreateStreamOnHGlobal(hMen, true, &pStream);
    if(ret == S_OK){
        //将文件保存到内存流里面
        image.Save(pStream. ::Gdiplus::ImageFormatPNG);
        //将流指针放到开头
        LARGE_INTEGER LG = {0};
        pStream->Seek(&LG, STREAM_SEEK_SET, NULL);
        //接下来获取流指针
        char* pdata = (char*)GlobalLock(hMen);
        //获取流长度
        int len = GlobalSize(hMen);
    }

    //发送本地屏幕
    return 0;
}
